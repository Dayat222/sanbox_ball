int hooked_open(const char* pathname, int flags, ...) {
    mode_t mode = 0;
    if (flags & O_CREAT) {
        va_list args;
        va_start(args, flags);
        mode = static_cast<mode_t>(va_arg(args, int));
        va_end(args);
    }

    if (!pathname) {
        errno = EFAULT;
        return -1;
    }

    std::string resolved = VfsCore::getInstance().resolvePath(pathname);
    if (resolved != pathname) {
        ALOGD("open() redirected: %s -> %s", pathname, resolved.c_str());
    }

    if (!g_orig_open) {
        g_orig_open = reinterpret_cast<orig_open_t>(dlsym(RTLD_NEXT, "open"));
    }

    if (g_orig_open) {
        return (flags & O_CREAT) ? g_orig_open(resolved.c_str(), flags, mode)
                                 : g_orig_open(resolved.c_str(), flags);
    }
    return -1;
}

static bool is_page_strictly_relro(uintptr_t page_start, size_t page_size, ElfW(Addr) base, const ElfW(Phdr)* phdr, ElfW(Half) phnum) {
    uintptr_t page_end = page_start + page_size;
    for (ElfW(Half) i = 0; i < phnum; ++i) {
        if (phdr[i].p_type == PT_GNU_RELRO) {
            uintptr_t relro_start = base + phdr[i].p_vaddr;
            uintptr_t relro_end = relro_start + phdr[i].p_memsz;
            // The entire page must be strictly within the PT_GNU_RELRO boundaries.
            // If the page extends beyond relro_end, it overlaps with writable .data / .bss
            // and locking it to PROT_READ causes SIGSEGV when that data is written!
            if (page_start >= relro_start && page_end <= relro_end) {
                return true;
            }
        }
    }
    return false;
}

static int plt_hook_callback(struct dl_phdr_info* info, size_t /* size */, void* data) {
    if (!info || !info->dlpi_name || !data) return 0;
    const char* lib_name = info->dlpi_name;

    // Skip self, libc, libdl, and core Android system/ART runtime libraries.
    // Hooking core ART/system libraries causes internal VM deadlocks or fatal aborts!
    if (strstr(lib_name, "libstealth_engine.so") != nullptr ||
        strstr(lib_name, "libc.so") != nullptr ||
        strstr(lib_name, "libdl.so") != nullptr ||
        strstr(lib_name, "libm.so") != nullptr ||
        strstr(lib_name, "libart.so") != nullptr ||
        strstr(lib_name, "libart-compiler.so") != nullptr ||
        strstr(lib_name, "libopenjdk.so") != nullptr ||
        strstr(lib_name, "libopenjdkjvm.so") != nullptr ||
        strstr(lib_name, "libandroid_runtime.so") != nullptr ||
        strstr(lib_name, "liblog.so") != nullptr ||
        strstr(lib_name, "libbinder.so") != nullptr ||
        strstr(lib_name, "libutils.so") != nullptr) {
        return 0;
    }

    ElfW(Addr) base = info->dlpi_addr;
    const ElfW(Phdr)* phdr = info->dlpi_phdr;
    ElfW(Half) phnum = info->dlpi_phnum;

    // Find PT_DYNAMIC
    const ElfW(Dyn)* dynamic = nullptr;
    for (ElfW(Half) i = 0; i < phnum; ++i) {
        if (phdr[i].p_type == PT_DYNAMIC) {
            dynamic = reinterpret_cast<const ElfW(Dyn)*>(base + phdr[i].p_vaddr);
            break;
        }
    }
    if (!dynamic) return 0;

    const char* strtab = nullptr;
    const ElfW(Sym)* symtab = nullptr;
    const void* jmprel = nullptr;
    size_t plt_rel_sz = 0;
    ElfW(Word) plt_rel_type = 0;

    for (const ElfW(Dyn)* d = dynamic; d->d_tag != DT_NULL; ++d) {
        switch (d->d_tag) {
            case DT_STRTAB: {
                ElfW(Addr) ptr = d->d_un.d_ptr;
                strtab = reinterpret_cast<const char*>(ptr < base ? base + ptr : ptr);
                break;
            }
            case DT_SYMTAB: {
                ElfW(Addr) ptr = d->d_un.d_ptr;
                symtab = reinterpret_cast<const ElfW(Sym)*>(ptr < base ? base + ptr : ptr);
                break;
            }
            case DT_JMPREL: {
                ElfW(Addr) ptr = d->d_un.d_ptr;
                jmprel = reinterpret_cast<const void*>(ptr < base ? base + ptr : ptr);
                break;
            }
            case DT_PLTRELSZ:
                plt_rel_sz = d->d_un.d_val;
                break;
            case DT_PLTREL:
                plt_rel_type = d->d_un.d_val;
                break;
            default:
                break;
        }
    }

    if (!strtab || !symtab || !jmprel || plt_rel_sz == 0) {
        return 0;
    }

    long page_size = sysconf(_SC_PAGESIZE);
    if (page_size <= 0) page_size = 4096;

    auto* targets = reinterpret_cast<const std::vector<PltHookTarget>*>(data);

    if (plt_rel_type == DT_RELA) {
        const auto* rela = reinterpret_cast<const ElfW(Rela)*>(jmprel);
        size_t count = plt_rel_sz / sizeof(ElfW(Rela));
        for (size_t i = 0; i < count; ++i) {
            size_t sym_idx = STEALTH_ELF_R_SYM(rela[i].r_info);
            const char* name = strtab + symtab[sym_idx].st_name;
            for (const auto& target : *targets) {
                if (strcmp(name, target.symbol_name) == 0) {
                    void** got_entry = reinterpret_cast<void**>(base + rela[i].r_offset);
                    if (*got_entry != target.hook_func) {
                        void* page_start = reinterpret_cast<void*>(
                            reinterpret_cast<uintptr_t>(got_entry) & ~(page_size - 1));
                        if (mprotect(page_start, page_size, PROT_READ | PROT_WRITE) == 0) {
                            if (target.orig_func_ptr && !*(target.orig_func_ptr)) {
                                *(target.orig_func_ptr) = *got_entry;
                            }
                            *got_entry = target.hook_func;
                            __builtin___clear_cache(
                                reinterpret_cast<char*>(got_entry),
                                reinterpret_cast<char*>(got_entry) + sizeof(void*)
                            );
                            ALOGI("PLT Hooked %s in %s (GOT: %p)", name, lib_name, got_entry);

                            // SEC-03: Restore PROT_READ strictly for pages fully inside PT_GNU_RELRO
                            if (is_page_strictly_relro(reinterpret_cast<uintptr_t>(page_start), page_size, base, phdr, phnum)) {
                                mprotect(page_start, page_size, PROT_READ);
                            }
                        }
                    }
                    break;
                }
            }
        }
    } else if (plt_rel_type == DT_REL) {
        const auto* rel = reinterpret_cast<const ElfW(Rel)*>(jmprel);
        size_t count = plt_rel_sz / sizeof(ElfW(Rel));
        for (size_t i = 0; i < count; ++i) {
            size_t sym_idx = STEALTH_ELF_R_SYM(rel[i].r_info);
            const char* name = strtab + symtab[sym_idx].st_name;
            for (const auto& target : *targets) {
                if (strcmp(name, target.symbol_name) == 0) {
                    void** got_entry = reinterpret_cast<void**>(base + rel[i].r_offset);
                    if (*got_entry != target.hook_func) {
                        void* page_start = reinterpret_cast<void*>(
                            reinterpret_cast<uintptr_t>(got_entry) & ~(page_size - 1));
                        if (mprotect(page_start, page_size, PROT_READ | PROT_WRITE) == 0) {
                            if (target.orig_func_ptr && !*(target.orig_func_ptr)) {
                                *(target.orig_func_ptr) = *got_entry;
                            }
                            *got_entry = target.hook_func;
                            __builtin___clear_cache(
                                reinterpret_cast<char*>(got_entry),
                                reinterpret_cast<char*>(got_entry) + sizeof(void*)
                            );
                            ALOGI("PLT Hooked %s in %s (GOT: %p)", name, lib_name, got_entry);

                            // SEC-03: Restore PROT_READ strictly for pages fully inside PT_GNU_RELRO
                            if (is_page_strictly_relro(reinterpret_cast<uintptr_t>(page_start), page_size, base, phdr, phnum)) {
                                mprotect(page_start, page_size, PROT_READ);
                            }
                        }
                    }
                    break;
                }
            }
        }
    }

    return 0;
}

int stealth_apply_plt_hooks() {
    auto targets = get_hook_targets();
    dl_iterate_phdr(plt_hook_callback, &targets);
    ALOGI("Stealth Hook Engine: PLT/GOT scan and interception completed.");
    return 0;
}

int stealth_install_hooks() {
    if (g_hooks_installed.load(std::memory_order_acquire)) {
        stealth_apply_plt_hooks();
        return 0;
    }

    void* libc = dlopen("libc.so", RTLD_NOW);
    if (libc) {
        g_orig_open   = reinterpret_cast<orig_open_t>(dlsym(libc, "open"));
        g_orig_openat = reinterpret_cast<orig_openat_t>(dlsym(libc, "openat"));
        g_orig_fopen  = reinterpret_cast<orig_fopen_t>(dlsym(libc, "fopen"));
        g_orig_access = reinterpret_cast<orig_access_t>(dlsym(libc, "access"));
        g_orig_stat   = reinterpret_cast<orig_stat_t>(dlsym(libc, "stat"));
        g_orig_lstat  = reinterpret_cast<orig_lstat_t>(dlsym(libc, "lstat"));
        g_orig_ptrace = reinterpret_cast<orig_ptrace_t>(dlsym(libc, "ptrace"));
    }

    // Apply PLT/GOT memory table redirection
    stealth_apply_plt_hooks();

    g_hooks_installed.store(1, std::memory_order_release);
    ALOGI("Stealth Hook Engine: Libc interception hooks installed.");
    return 0;
}

    g_hooks_installed.store(0, std::memory_order_release);
    ALOGI("Stealth Hook Engine: Libc interception hooks uninstalled.");
    return 0;
}

static void ensureDirectory(const std::string& path) {
    size_t pos = 0;
    while ((pos = path.find('/', pos + 1)) != std::string::npos) {
        std::string sub = path.substr(0, pos);
        if (!sub.empty()) {
            mkdir(sub.c_str(), 0755);
        }
    }
    mkdir(path.c_str(), 0755);
}

/**
 * Lexically normalizes a path by resolving '.' and '..' segments without requiring file existence.
 * Prevents Directory Traversal (SEC-02).
 */
static std::string normalizeLexicalPath(const std::string& rawPath) {
    if (rawPath.empty()) return "";
    bool isAbsolute = (rawPath[0] == '/');
    std::vector<std::string> parts;
    std::stringstream ss(rawPath);
    std::string item;
    while (std::getline(ss, item, '/')) {
        if (item.empty() || item == ".") continue;
        if (item == "..") {
            if (!parts.empty()) parts.pop_back();
        } else {
            parts.push_back(item);
        }
    }
    std::string result = isAbsolute ? "/" : "";
    for (size_t i = 0; i < parts.size(); ++i) {
        result += parts[i];
        if (i + 1 < parts.size()) result += "/";
    }
    return result;
}

/**
 * Checks specifically whether the target path targets /proc/[pid]/maps.
 * Avoids false positive interceptions of application files containing '/maps' (SEC-02).
 */
static bool isProcMapsPath(const std::string& path) {
    if (path == "/proc/self/maps" || path == "/proc/thread-self/maps") return true;
    if (path.rfind("/proc/", 0) == 0 && path.length() > 11) {
        return path.compare(path.length() - 5, 5, "/maps") == 0;
    }
    return false;
}

void VfsCore::init(const std::string& virtualRootDir, const std::string& targetPkg,
                   const std::string& fakeMac, const std::string& fakeSerial) {
    std::lock_guard<std::mutex> lock(mutex_);
    virtualRootDir_ = virtualRootDir;
    targetPkg_ = targetPkg;
    fakeMac_ = fakeMac.empty() ? "02:00:00:00:00:00" : fakeMac;
    fakeSerial_ = fakeSerial.empty() ? "RF8M1234XYZ" : fakeSerial;

    ensureDirectory(virtualRootDir_);
    std::string fakeDir = virtualRootDir_ + "/fake_sys";
    ensureDirectory(fakeDir);

    // Clean up old temporary proc_maps.* files from previous sessions
    DIR* dir = opendir(fakeDir.c_str());
    if (dir) {
        struct dirent* entry;
        while ((entry = readdir(dir)) != nullptr) {
            if (strncmp(entry->d_name, "proc_maps", 9) == 0) {
                std::string filePath = fakeDir + "/" + entry->d_name;
                unlink(filePath.c_str());
            }
        }
        closedir(dir);
    }

    fakeCpuInfoPath_ = fakeDir + "/cpuinfo";
    fakeVersionPath_ = fakeDir + "/version";
    fakeMacPath_ = fakeDir + "/mac_address";
    fakeStatusPath_ = fakeDir + "/proc_status";
    sanitizedMapsPath_ = fakeDir + "/proc_maps";

    generateSpoofedFiles();

    // Map sensitive proc and sys paths
    exactMappings_["/proc/cpuinfo"] = fakeCpuInfoPath_;
    exactMappings_["/proc/version"] = fakeVersionPath_;
    exactMappings_["/sys/class/net/wlan0/address"] = fakeMacPath_;
    exactMappings_["/sys/class/net/eth0/address"] = fakeMacPath_;
    exactMappings_["/proc/self/status"] = fakeStatusPath_;

    // Mount namespace & directory mapping:
    // When app requests /data/data/<pkg> or /data/user/0/<pkg>, redirect to container isolated path
    if (!targetPkg_.empty()) {
        std::string virtualDataDir = virtualRootDir_ + "/data/" + targetPkg_;
        ensureDirectory(virtualDataDir);
        prefixMappings_.push_back({"/data/data/" + targetPkg_, virtualDataDir});
        prefixMappings_.push_back({"/data/user/0/" + targetPkg_, virtualDataDir});
    }

    enabled_.store(true, std::memory_order_release);
    ALOGI("VfsCore initialized successfully for package: %s, root: %s",
          targetPkg_.c_str(), virtualRootDir_.c_str());
}

void VfsCore::addPathMapping(const std::string& virtualPath, const std::string& realPath) {

    exactMappings_[virtualPath] = realPath;

std::string VfsCore::resolvePath(const char* originalPath) {
    if (!originalPath || !enabled_.load(std::memory_order_acquire)) {
        return originalPath ? std::string(originalPath) : "";
    }

    std::string normPath = normalizeLexicalPath(originalPath);
    if (normPath.empty()) {
        return originalPath ? std::string(originalPath) : "";
    }

    // Targeted interception for /proc/[pid]/maps (SEC-02)
    if (mapsFilterEnabled_.load(std::memory_order_acquire) && isProcMapsPath(normPath)) {
        std::lock_guard<std::mutex> lock(mutex_);
        std::string sanitized = sanitizeMapsFile();
        if (!sanitized.empty()) {
            return sanitized;
        }
    }

    std::lock_guard<std::mutex> lock(mutex_);

    // SEC-02: Protect host container data directory from sandbox escape!
    // Block target container from reading com.sandbox.ball host data outside virtualRootDir_
    // ONLY enforce when targetPkg_ is set (meaning a guest application is running).
    if (!targetPkg_.empty()) {
        const std::string hostData1 = "/data/data/com.sandbox.ball";
        const std::string hostData2 = "/data/user/0/com.sandbox.ball";
        bool pointsToHost = false;
        if ((normPath.rfind(hostData1, 0) == 0 && (normPath.length() == hostData1.length() || normPath[hostData1.length()] == '/')) ||
            (normPath.rfind(hostData2, 0) == 0 && (normPath.length() == hostData2.length() || normPath[hostData2.length()] == '/'))) {
            pointsToHost = true;
        }

        if (pointsToHost) {
            if (virtualRootDir_.empty() || normPath.rfind(virtualRootDir_, 0) != 0) {
                // Allow host native libraries and runtime cache needed by the container process
                if (normPath.find("/lib") == std::string::npos && normPath.find("/code_cache") == std::string::npos) {
                    ALOGW("VfsCore: Blocked attempted sandbox escape to host path: %s", normPath.c_str());
                    return "/dev/null";
                }
            }
        }
    }

    // Check exact matches
    auto it = exactMappings_.find(normPath);
    if (it != exactMappings_.end()) {
        return it->second;
    }

    // Check prefix matches with directory boundary check (SEC-02)
    for (const auto& pair : prefixMappings_) {
        const std::string& prefix = pair.first;
        if (normPath.rfind(prefix, 0) == 0 &&
            (normPath.length() == prefix.length() || normPath[prefix.length()] == '/')) {
            std::string sub = normPath.substr(prefix.length());
            return pair.second + sub;
        }
    }

    return normPath;
}

bool VfsCore::shouldRedirect(const char* originalPath) const {
    if (!originalPath || !enabled_.load(std::memory_order_acquire)) return false;

    std::string normPath = normalizeLexicalPath(originalPath);
    if (normPath.empty()) return false;

    if (mapsFilterEnabled_.load(std::memory_order_acquire) && isProcMapsPath(normPath)) {
        return true;
    }

    std::lock_guard<std::mutex> lock(mutex_);

    if (!targetPkg_.empty()) {
        const std::string hostData1 = "/data/data/com.sandbox.ball";
        const std::string hostData2 = "/data/user/0/com.sandbox.ball";
        if ((normPath.rfind(hostData1, 0) == 0 && (normPath.length() == hostData1.length() || normPath[hostData1.length()] == '/')) ||
            (normPath.rfind(hostData2, 0) == 0 && (normPath.length() == hostData2.length() || normPath[hostData2.length()] == '/'))) {
            if (virtualRootDir_.empty() || normPath.rfind(virtualRootDir_, 0) != 0) {
                if (normPath.find("/lib") == std::string::npos && normPath.find("/code_cache") == std::string::npos) {
                    return true;
                }
            }
        }
    }

    if (exactMappings_.find(normPath) != exactMappings_.end()) {
        return true;
    }

    for (const auto& pair : prefixMappings_) {
        const std::string& prefix = pair.first;
        if (normPath.rfind(prefix, 0) == 0 &&
            (normPath.length() == prefix.length() || normPath[prefix.length()] == '/')) {
            return true;
        }
    }
    return false;
}

std::string VfsCore::sanitizeMapsFile() {
    std::ifstream src("/proc/self/maps");
    if (!src.is_open()) {
        return "";
    }

    // Thread-safe atomic output per thread to prevent concurrent race conditions
    std::string tidPath = sanitizedMapsPath_ + "." + std::to_string(gettid());
    std::ofstream dst(tidPath, std::ios::trunc);
    if (!dst.is_open()) {
        src.close();
        return "";
    }

    std::string line;
    while (std::getline(src, line)) {
        // Strip references to sandbox engine, hooking libs, xposed, frida
        if (line.find("libstealth_engine.so") != std::string::npos ||
            line.find("com.sandbox.ball") != std::string::npos ||
            line.find("frida") != std::string::npos ||
            line.find("xposed") != std::string::npos ||
            line.find("substrate") != std::string::npos ||
            line.find("dexrepair") != std::string::npos) {
            continue; // Skip sensitive line
        }
        dst << line << "\n";
    }

    src.close();
    dst.close();
    return tidPath;
}

        val type = object : TypeToken<List<ClonedAppInfo>>() {}.type
        return try {
            gson.fromJson(json, type) ?: emptyList()
        } catch (_: Exception) {
            emptyList()
        }
    }

    private fun saveClonedApps(list: List<ClonedAppInfo>) {
        prefs.edit().putString(KEY_CLONED_APPS, gson.toJson(list)).apply()
    }

        return getClonedApps().any { it.packageName == packageName }
    }

    /**
     * PRD Core Feature: Kloning Aplikasi
     * Copies APK from OS into sandbox container directory without root.
     */
    fun cloneApp(packageName: String): ClonedAppInfo {
        val pm = context.packageManager
        val pkgInfo = pm.getPackageInfo(packageName, PackageManager.GET_ACTIVITIES)
        val appInfo = pkgInfo.applicationInfo ?: throw IllegalArgumentException("ApplicationInfo not found")

        val appName = pm.getApplicationLabel(appInfo).toString()
        val sourceApk = File(appInfo.sourceDir)
        val containerDir = vfs.getAppContainerDir(packageName)
        val targetApk = File(containerDir, "base.apk")

        // Copy APK to isolated sandbox container
        copyFile(sourceApk, targetApk)

        // Initialize virtual directories
        vfs.getAppDataDir(packageName)
        vfs.registerAppVfs(packageName)

        val activeProfile = profileManager.getActiveProfile()

        val clonedApp = ClonedAppInfo(
            packageName = packageName,
            appName = appName,
            versionName = pkgInfo.versionName ?: "1.0",
            versionCode = if (android.os.Build.VERSION.SDK_INT >= android.os.Build.VERSION_CODES.P) {
                pkgInfo.longVersionCode
            } else {
                @Suppress("DEPRECATION")
                pkgInfo.versionCode.toLong()
            },
            sourceApkPath = targetApk.absolutePath,
            containerDir = containerDir.absolutePath,
            cloneDate = System.currentTimeMillis(),
            profileId = activeProfile?.profileId ?: "",
            profileName = activeProfile?.profileName ?: "Default Stealth Profile",
            isStealthShieldEnabled = true
        )

        val index = currentList.indexOfFirst { it.packageName == packageName }

            currentList[index] = clonedApp
        } else {
            currentList.add(clonedApp)

                profileName = newProfile.profileName

package com.sandbox.ball.core.container

import android.os.Build
import android.util.Log
import java.io.File
import java.io.FileOutputStream
import java.util.zip.ZipFile

object NativeLibExtractor {
    private const val TAG = "NativeLibExtractor"

    /**
     * Validates that an entry inside the ZIP archive does not escape the destination directory.
     * Prevents Zip Slip (Arbitrary File Overwrite) attacks (SEC-01).
     */
    fun validateEntry(entryName: String, prefix: String, targetLibDir: File): File {
        if (!entryName.startsWith(prefix)) {
            throw IllegalArgumentException("Entry '$entryName' does not start with prefix '$prefix'")
        }
        val soName = entryName.substring(prefix.length)
        if (soName.contains("/") || soName.contains("\\") || soName.contains("..")) {
            throw SecurityException("Zip Slip attack detected on entry: $entryName")
        }

        val canonicalDestDir = targetLibDir.canonicalPath + File.separator
        val outFile = File(targetLibDir, soName)
        val canonicalOutPath = outFile.canonicalPath

        if (!canonicalOutPath.startsWith(canonicalDestDir)) {
            throw SecurityException("Zip Slip attack detected on entry: $entryName")
        }

        return outFile
    }

    fun extractNativeLibs(apkFile: File, targetLibDir: File) {
        if (!apkFile.exists()) return
        if (!targetLibDir.exists()) targetLibDir.mkdirs()

        try {
            ZipFile(apkFile).use { zip ->
                val entries = zip.entries().toList()
                val supportedAbis = Build.SUPPORTED_ABIS

                // Find the best matching ABI present in the APK
                var selectedAbi: String? = null
                for (abi in supportedAbis) {
                    val hasAbi = entries.any { it.name.startsWith("lib/$abi/") && it.name.endsWith(".so") }
                    if (hasAbi) {
                        selectedAbi = abi
                        break
                    }
                }

                if (selectedAbi == null) {
                    Log.i(TAG, "No native libraries found for supported ABIs in ${apkFile.name}")
                    return
                }

                Log.i(TAG, "Extracting native libraries for ABI: $selectedAbi to ${targetLibDir.absolutePath}")
                val prefix = "lib/$selectedAbi/"

                for (entry in entries) {
                    if (entry.name.startsWith(prefix) && entry.name.endsWith(".so")) {
                        try {
                            val outFile = validateEntry(entry.name, prefix, targetLibDir)

                            // Only extract if doesn't exist or size mismatch
                            if (!outFile.exists() || outFile.length() != entry.size) {
                                zip.getInputStream(entry).use { inStream ->
                                    FileOutputStream(outFile).use { outStream ->
                                        inStream.copyTo(outStream)
                                    }
                                }
                                Log.d(TAG, "Extracted native lib: ${outFile.name} (${outFile.length()} bytes)")
                            }
                        } catch (se: SecurityException) {
                            Log.w(TAG, "Blocked suspicious zip entry: ${entry.name} - ${se.message}")
                        }
                    }
                }
            }
        } catch (e: Exception) {
            Log.e(TAG, "Failed extracting native libraries from ${apkFile.name}: ${e.message}", e)
        }
    }
}

    private class PackageManagerInvocationHandler(private val realObject: Any) : InvocationHandler {
        override fun invoke(proxy: Any?, method: Method, args: Array<out Any>?): Any? {
            try {
                val methodName = method.name

                // Intercept Play Store installer source (Anti-Shadowban)
                if (methodName == "getInstallerPackageName") {
                    val pkg = args?.getOrNull(0) as? String ?: ""
                    return PackageManagerHook.getInstallerPackageName(pkg)
                }

                if (methodName == "getInstallSourceInfo") {
                    val fake = createFakeInstallSourceInfo()
                    if (fake != null) return fake
                    return invokeReal(realObject, method, args)
                }

                // Filter package visibility
                if (methodName == "getPackageInfo") {
                    val pkgName = args?.getOrNull(0) as? String
                    if (pkgName != null && !PackageManagerHook.isPackageVisible(pkgName)) {
                        // In AIDL IPC, returning null signifies package not found.
                        // Never throw NameNotFoundException directly via reflection!
                        return null
                    }
                }

                // Filter package lists
                if (methodName == "getInstalledPackages") {
                    val raw = invokeReal(realObject, method, args) ?: return null
                    return filterParceledOrList(raw) { list ->
                        @Suppress("UNCHECKED_CAST")
                        PackageManagerHook.filterPackageInfoList(list as? List<PackageInfo>)
                    }
                }

                if (methodName == "getInstalledApplications") {
                    val raw = invokeReal(realObject, method, args) ?: return null
                    return filterParceledOrList(raw) { list ->
                        @Suppress("UNCHECKED_CAST")
                        PackageManagerHook.filterApplicationInfoList(list as? List<ApplicationInfo>)
                    }
                }

                if (methodName == "queryIntentActivities") {
                    val raw = invokeReal(realObject, method, args) ?: return null
                    return filterParceledOrList(raw) { list ->
                        @Suppress("UNCHECKED_CAST")
                        PackageManagerHook.filterResolveInfoList(list as? List<ResolveInfo>)
                    }
                }

                return invokeReal(realObject, method, args)
            } catch (t: Throwable) {
                Log.w(TAG, "PackageManager proxy fallback on ${method.name}: ${t.message}")
                return invokeReal(realObject, method, args)
            }
        }

        private fun filterParceledOrList(raw: Any, filterAction: (List<*>) -> List<*>): Any {
            return try {
                val clazz = raw.javaClass
                if (clazz.name.contains("ParceledListSlice")) {
                    val getListMethod = clazz.getMethod("getList")
                    val originalList = getListMethod.invoke(raw) as? List<*> ?: emptyList<Any>()
                    val filtered = filterAction(originalList)
                    val constructor = clazz.getConstructor(List::class.java)
                    constructor.newInstance(filtered)
                } else if (raw is List<*>) {
                    filterAction(raw)
                } else {
                    raw
                }
            } catch (_: Exception) {
                raw
            }
        }

        private fun createFakeInstallSourceInfo(): Any? {
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.R) {
                return try {
                    val clazz = Class.forName("android.content.pm.InstallSourceInfo")
                    for (ctor in clazz.declaredConstructors) {
                        ctor.isAccessible = true
                        val paramTypes = ctor.parameterTypes
                        val args = arrayOfNulls<Any>(paramTypes.size)
                        for (i in paramTypes.indices) {
                            if (paramTypes[i] == String::class.java) {
                                args[i] = "com.android.vending"
                            }
                        }
                        return ctor.newInstance(*args)
                    }
                    null
                } catch (_: Exception) {
                    null
                }
            }
            return null
        }
    }

    private fun invokeReal(target: Any, method: Method, args: Array<out Any>?): Any? {
        return try {
            if (args == null) {
                method.invoke(target)
            } else {
                method.invoke(target, *args)
            }
        } catch (e: java.lang.reflect.InvocationTargetException) {
            throw e.targetException ?: e
        }
    }

        if (!isVisibilityHidingEnabled || originalList == null || targetPackageName == null) {
            return originalList ?: emptyList()
        }

        return originalList.filter { pkg ->
            val name = pkg.packageName
            val isTarget = name == targetPackageName
            val appInfo = pkg.applicationInfo
            val isSystem = appInfo != null && ((appInfo.flags and ApplicationInfo.FLAG_SYSTEM) != 0)
            val isBlacklisted = BLACKLISTED_PACKAGES.contains(name) || name.contains("sandbox")

            // Cloned app only sees itself and standard system packages, never sandbox host or root tools
            (isTarget || isSystem) && !isBlacklisted
        }
    }

        if (!isVisibilityHidingEnabled || originalList == null || targetPackageName == null) {
            return originalList ?: emptyList()
        }

        return originalList.filter { app ->
            val name = app.packageName
            val isTarget = name == targetPackageName
            val isSystem = (app.flags and ApplicationInfo.FLAG_SYSTEM) != 0
            val isBlacklisted = BLACKLISTED_PACKAGES.contains(name) || name.contains("sandbox")

            (isTarget || isSystem) && !isBlacklisted
        }
    }

    fun filterResolveInfoList(originalList: List<ResolveInfo>?): List<ResolveInfo> {
        if (!isVisibilityHidingEnabled || originalList == null || targetPackageName == null) {
            return originalList ?: emptyList()
        }

        return originalList.filter { resolve ->

            !BLACKLISTED_PACKAGES.contains(pkgName) && !pkgName.contains("sandbox")
        }
    }

        if (targetPackageName == null) return true

    private const val GITHUB_REPO_RELEASES_API = "https://api.github.com/repos/Dayat222/sanbox_ball/releases"
    private const val GITHUB_REPO_LATEST_API = "https://api.github.com/repos/Dayat222/sanbox_ball/releases/latest"

    /**
     * Verifies that the update URL uses HTTPS and targets official GitHub domains (SEC-04).
     */
    fun isTrustedUpdateUrl(url: String): Boolean {
        return try {
            val parsed = URL(url)
            val scheme = parsed.protocol
            val host = parsed.host?.lowercase() ?: ""
            scheme.equals("https", ignoreCase = true) && (
                host == "github.com" ||
                host == "objects.githubusercontent.com" ||
                host.endsWith(".github.com") ||
                host.endsWith(".githubusercontent.com")
            )
        } catch (e: Exception) {
            false
        }
    }

    /**
     * Calculates the SHA-256 checksum of a file.
     */
    fun calculateSha256(file: File): String {
        val digest = MessageDigest.getInstance("SHA-256")
        file.inputStream().use { fis ->
            val buffer = ByteArray(8192)
            var bytesRead: Int
            while (fis.read(buffer).also { bytesRead = it } != -1) {
                digest.update(buffer, 0, bytesRead)
            }
        }
        return digest.digest().joinToString("") { "%02x".format(it) }
    }

    /**
     * Compares the computed SHA-256 against an expected checksum in constant time.
     */
    fun verifySha256(file: File, expectedSha256: String): Boolean {
        if (!file.exists()) return false
        if (expectedSha256.isBlank()) return true
        val actualSha256 = calculateSha256(file)
        val expectedBytes = expectedSha256.trim().lowercase().toByteArray(Charsets.UTF_8)
        val actualBytes = actualSha256.toByteArray(Charsets.UTF_8)
        return MessageDigest.isEqual(expectedBytes, actualBytes)
    }

    /**
     * Verifies the integrity of a downloaded APK:
     * 1. Checksum matching (constant-time MessageDigest comparison)
     * 2. Package name validation
     * 3. Signing certificate verification against currently installed host
     */
    fun verifyDownloadedApk(
        context: Context,
        apkFile: File,
        expectedSha256: String = ""
    ): Boolean {
        if (!apkFile.exists()) return false

        // 1. Checksum verification
        if (expectedSha256.isNotBlank() && !verifySha256(apkFile, expectedSha256)) {
            Log.e(TAG, "SHA-256 mismatch for APK: ${apkFile.name}")
            apkFile.delete()
            return false
        }

        // 2. Package name and certificate verification
        try {
            val pm = context.packageManager
            val archiveInfo = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.P) {
                pm.getPackageArchiveInfo(apkFile.absolutePath, PackageManager.GET_SIGNING_CERTIFICATES)
            } else {
                @Suppress("DEPRECATION")
                pm.getPackageArchiveInfo(apkFile.absolutePath, PackageManager.GET_SIGNATURES)
            }

            if (archiveInfo == null) {
                Log.e(TAG, "Could not parse APK archive info: ${apkFile.name}")
                apkFile.delete()
                return false
            }

            if (archiveInfo.packageName != context.packageName) {
                Log.e(TAG, "Package name mismatch! Expected: ${context.packageName}, Found: ${archiveInfo.packageName}")
                apkFile.delete()
                return false
            }

            val installedInfo = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.P) {
                pm.getPackageInfo(context.packageName, PackageManager.GET_SIGNING_CERTIFICATES)
            } else {
                @Suppress("DEPRECATION")
                pm.getPackageInfo(context.packageName, PackageManager.GET_SIGNATURES)
            }

            val signaturesMatch = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.P) {
                val curSign = installedInfo.signingInfo
                val archSign = archiveInfo.signingInfo
                if (curSign != null && archSign != null) {
                    val curSigs = if (curSign.hasMultipleSigners()) curSign.apkContentsSigners else curSign.signingCertificateHistory
                    val archSigs = if (archSign.hasMultipleSigners()) archSign.apkContentsSigners else archSign.signingCertificateHistory
                    curSigs != null && archSigs != null && curSigs.any { cur ->
                        archSigs.any { arch -> MessageDigest.isEqual(cur.toByteArray(), arch.toByteArray()) }
                    }
                } else false
            } else {
                @Suppress("DEPRECATION")
                val curSigs = installedInfo.signatures
                @Suppress("DEPRECATION")
                val archSigs = archiveInfo.signatures
                if (curSigs != null && archSigs != null) {
                    curSigs.any { cur ->
                        archSigs.any { arch -> MessageDigest.isEqual(cur.toByteArray(), arch.toByteArray()) }
                    }
                } else false
            }

            if (!signaturesMatch) {
                Log.e(TAG, "Signature mismatch between installed app and downloaded APK!")
                apkFile.delete()
                return false
            }
        } catch (e: Exception) {
            Log.e(TAG, "Failed verifying downloaded APK signatures: ${e.message}", e)
            apkFile.delete()
            return false
        }

        return true
    }

    suspend fun checkForUpdate(): AppUpdateInfo? = withContext(Dispatchers.IO) {
        try {
            val json = fetchReleaseJson(GITHUB_REPO_RELEASES_API) 
                ?: fetchReleaseJson(GITHUB_REPO_LATEST_API)
                ?: return@withContext null

            val tagName = json.get("tag_name")?.asString ?: ""
            val releaseNotes = json.get("body")?.asString ?: "Perbaikan bug dan peningkatan sistem stealth."
            val assets = json.getAsJsonArray("assets")

            var downloadUrl = ""
            var apkName = "app-release.apk"
            var fileSize = 0L

            if (assets != null && assets.size() > 0) {
                // First pass: prioritize production release APK
                for (i in 0 until assets.size()) {
                    val asset = assets[i].asJsonObject
                    val name = asset.get("name")?.asString ?: ""
                    if (name.contains("release", ignoreCase = true) && name.endsWith(".apk", ignoreCase = true)) {
                        downloadUrl = asset.get("browser_download_url")?.asString ?: ""
                        apkName = name
                        fileSize = asset.get("size")?.asLong ?: 0L
                        break
                    }
                }
                // Second pass fallback: any available APK
                if (downloadUrl.isEmpty()) {
                    for (i in 0 until assets.size()) {
                        val asset = assets[i].asJsonObject
                        val name = asset.get("name")?.asString ?: ""
                        if (name.endsWith(".apk", ignoreCase = true)) {
                            downloadUrl = asset.get("browser_download_url")?.asString ?: ""
                            apkName = name
                            fileSize = asset.get("size")?.asLong ?: 0L
                            break
                        }
                    }
                }
            }

            val currentVersion = BuildConfig.VERSION_NAME
            val cleanTag = tagName.removePrefix("v").trim()
            val cleanCurrent = currentVersion.removePrefix("v").trim()

            // Extract optional SHA-256 hash from release notes (format: SHA256: <hash>)
            val shaRegex = Regex("""(?i)SHA-?256:\s*([a-fA-F0-9]{64})""")
            val shaMatch = shaRegex.find(releaseNotes)
            val sha256 = shaMatch?.groupValues?.get(1)?.lowercase() ?: ""

            AppUpdateInfo(

                latestVersion = tagName,

                apkFileName = apkName,

        }

    private fun fetchReleaseJson(apiUrl: String): com.google.gson.JsonObject? {
        return try {
            val url = URL(apiUrl)
            val connection = (url.openConnection() as HttpURLConnection).apply {
                requestMethod = "GET"
                connectTimeout = 8000
                readTimeout = 8000
                setRequestProperty("Accept", "application/vnd.github.v3+json")
                setRequestProperty("User-Agent", "StealthSandbox-App")
            }

            if (connection.responseCode != 200) {
                Log.w(TAG, "GitHub API $apiUrl returned code: ${connection.responseCode}")
                return null
            }

            val responseText = connection.inputStream.bufferedReader().use { it.readText() }
            val parsed = JsonParser.parseString(responseText)
            if (parsed.isJsonArray) {
                val array = parsed.asJsonArray
                if (array.size() == 0) return null
                var chosen: com.google.gson.JsonObject? = null
                for (i in 0 until array.size()) {
                    val item = array.get(i).asJsonObject
                    val assets = item.getAsJsonArray("assets")
                    if (assets != null) {
                        for (j in 0 until assets.size()) {
                            val assetName = assets.get(j).asJsonObject.get("name")?.asString ?: ""
                            if (assetName.endsWith(".apk", ignoreCase = true)) {
                                chosen = item
                                break
                            }
                        }
                    }
                    if (chosen != null) break
                }
                chosen ?: array.get(0).asJsonObject
            } else if (parsed.isJsonObject) {
                parsed.asJsonObject
            } else {
                null
            }
        } catch (e: Exception) {
            Log.w(TAG, "Error fetching from $apiUrl: ${e.message}")
            null
        }
    }

    private fun isNewerVersion(remote: String, current: String): Boolean {
        if (remote == current) return false
        val remoteParts = remote.split("-")[0].split(".").mapNotNull { it.toIntOrNull() }
        val currentParts = current.split("-")[0].split(".").mapNotNull { it.toIntOrNull() }

        val length = maxOf(remoteParts.size, currentParts.size)
        for (i in 0 until length) {
            val r = remoteParts.getOrElse(i) { 0 }
            val c = currentParts.getOrElse(i) { 0 }
            if (r > c) return true
            if (r < c) return false
        }
        return remote != current
    }

    /**
     * Downloads APK into secure internal cache directory (SEC-04).
     * Validates HTTPS scheme and prevents insecure HTTP downgrade redirects.
     */
    suspend fun downloadApk(
        context: Context,
        downloadUrl: String,
        onProgress: (Int) -> Unit
    ): File? = withContext(Dispatchers.IO) {
        try {
            if (!isTrustedUpdateUrl(downloadUrl)) {
                throw SecurityException("Insecure or untrusted download URL: $downloadUrl")
            }

            val updatesDir = File(context.cacheDir, "updates")
            if (!updatesDir.exists()) updatesDir.mkdirs()

            val apkFile = File(updatesDir, "stealth_update.apk")
            if (apkFile.exists()) apkFile.delete()

            var currentUrl = downloadUrl
            var connection: HttpURLConnection
            var redirectCount = 0
            val maxRedirects = 5

            while (true) {
                val url = URL(currentUrl)
                connection = (url.openConnection() as HttpURLConnection).apply {
                    connectTimeout = 15000
                    readTimeout = 15000
                    instanceFollowRedirects = false
                    setRequestProperty("User-Agent", "StealthSandbox-App")
                }

                val responseCode = connection.responseCode
                if (responseCode == HttpURLConnection.HTTP_MOVED_PERM ||
                    responseCode == HttpURLConnection.HTTP_MOVED_TEMP ||
                    responseCode == 307 || responseCode == 308) {

                    val newUrl = connection.getHeaderField("Location")
                        ?: throw SecurityException("Missing redirect Location header")

                    val resolvedNewUrl = URL(url, newUrl).toString()
                    if (!isTrustedUpdateUrl(resolvedNewUrl)) {
                        throw SecurityException("Insecure or untrusted redirect URL: $resolvedNewUrl")
                    }

                    connection.disconnect()
                    currentUrl = resolvedNewUrl
                    redirectCount++
                    if (redirectCount > maxRedirects) {
                        throw SecurityException("Too many redirects while downloading update")
                    }
                    continue
                } else if (responseCode == HttpURLConnection.HTTP_OK) {
                    break
                } else {
                    Log.w(TAG, "Download failed with HTTP status code: $responseCode")
                    return@withContext null
                }
            }

            val totalSize = connection.contentLength
            var downloaded = 0L

            connection.inputStream.use { input ->
                FileOutputStream(apkFile).use { output ->
                    val buffer = ByteArray(8 * 1024)
                    var read: Int
                    while (input.read(buffer).also { read = it } != -1) {
                        output.write(buffer, 0, read)
                        downloaded += read
                        if (totalSize > 0) {
                            val percent = ((downloaded * 100) / totalSize).toInt()
                            mainHandler.post { onProgress(percent) }
                        }
                    }
                    output.flush()
                }
            }

            apkFile
        } catch (e: Exception) {
            Log.e(TAG, "Error downloading APK: ${e.message}", e)
            null
        }
    }

    fun installApk(context: Context, apkFile: File, expectedSha256: String = "") {
        if (!apkFile.exists()) {
            Log.e(TAG, "APK file does not exist: ${apkFile.absolutePath}")
            return
        }

        if (!verifyDownloadedApk(context, apkFile, expectedSha256)) {
            Log.e(TAG, "APK verification failed; installation aborted.")
            return
        }

        // On Android 8.0+ (API 26+), check unknown apps install permission
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            if (!context.packageManager.canRequestPackageInstalls()) {
                val settingsIntent = Intent(Settings.ACTION_MANAGE_UNKNOWN_APP_SOURCES).apply {
                    data = Uri.parse("package:${context.packageName}")
                    addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
                }
                context.startActivity(settingsIntent)
                return
            }
        }

        val apkUri = FileProvider.getUriForFile(
            context,
            "${context.packageName}.fileprovider",
            apkFile
        )

        val installIntent = Intent(Intent.ACTION_VIEW).apply {
            setDataAndType(apkUri, "application/vnd.android.package-archive")
            addFlags(Intent.FLAG_GRANT_READ_URI_PERMISSION)
            addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
        }

        context.startActivity(installIntent)
    }
}

        // The main dashboard process ("com.sandbox.ball") must run unhindered to guarantee UI stability!

        }

import android.widget.ImageView
import android.widget.TextView
import androidx.recyclerview.widget.RecyclerView
import com.sandbox.ball.R
import com.sandbox.ball.model.InstalledAppInfo

class InstalledAppAdapter(
    private var apps: List<InstalledAppInfo>,
    private val onCloneClick: (InstalledAppInfo) -> Unit
) : RecyclerView.Adapter<InstalledAppAdapter.ViewHolder>() {

    class ViewHolder(view: View) : RecyclerView.ViewHolder(view) {
        val imgIcon: ImageView = view.findViewById(R.id.img_inst_app_icon)
        val txtName: TextView = view.findViewById(R.id.txt_inst_app_name)
        val txtPackage: TextView = view.findViewById(R.id.txt_inst_app_package)
        val txtVersion: TextView = view.findViewById(R.id.txt_inst_app_version)
        val btnClone: Button = view.findViewById(R.id.btn_inst_clone)
    }

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): ViewHolder {
        val view = LayoutInflater.from(parent.context)
            .inflate(R.layout.item_installed_app, parent, false)
        return ViewHolder(view)
    }

    override fun onBindViewHolder(holder: ViewHolder, position: Int) {
        val app = apps[position]
        holder.txtName.text = app.appName
        holder.txtPackage.text = app.packageName
        holder.txtVersion.text = "v${app.versionName}"

        try {
            val pm = holder.itemView.context.packageManager
            val icon: Drawable = pm.getApplicationIcon(app.packageName)
            holder.imgIcon.setImageDrawable(icon)
        } catch (_: PackageManager.NameNotFoundException) {
            holder.imgIcon.setImageResource(R.drawable.ic_stealth_shield)
        }

        if (app.isAlreadyCloned) {
            holder.btnClone.text = "CLONED"
            holder.btnClone.isEnabled = false
            holder.btnClone.alpha = 0.5f
        } else {
            holder.btnClone.text = "CLONE"
            holder.btnClone.isEnabled = true
            holder.btnClone.alpha = 1.0f
            holder.btnClone.setOnClickListener { onCloneClick(app) }
        }
    }

    override fun getItemCount(): Int = apps.size

    fun updateData(newApps: List<InstalledAppInfo>) {
        this.apps = newApps
        notifyDataSetChanged()
    }
}

package com.sandbox.ball.ui

import android.annotation.SuppressLint
import android.content.pm.ApplicationInfo
import android.content.pm.PackageManager
import android.os.Bundle
import android.text.Editable
import android.text.TextWatcher
import android.view.View

import android.widget.TextView

import androidx.lifecycle.lifecycleScope
import androidx.recyclerview.widget.LinearLayoutManager
import androidx.recyclerview.widget.RecyclerView
import com.sandbox.ball.R
import com.sandbox.ball.core.container.AppContainer
import com.sandbox.ball.model.InstalledAppInfo
import com.sandbox.ball.ui.adapter.InstalledAppAdapter
import com.sandbox.ball.ui.view.StealthToast
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.launch
import kotlinx.coroutines.withContext

class AppClonerActivity : AppCompatActivity() {

    private lateinit var appContainer: AppContainer
    private lateinit var rvInstalledApps: RecyclerView
    private lateinit var adapter: InstalledAppAdapter
    private lateinit var progressBar: ProgressBar
    private lateinit var etSearch: EditText
    private lateinit var txtAppCount: TextView

    private var allApps: List<InstalledAppInfo> = emptyList()

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_app_cloner)

        appContainer = AppContainer.getInstance(this)

        rvInstalledApps = findViewById(R.id.rv_installed_apps)
        progressBar = findViewById(R.id.progress_cloner)
        etSearch = findViewById(R.id.et_search_apps)
        txtAppCount = findViewById(R.id.txt_installed_count)

        setupRecyclerView()
        setupSearch()
        loadInstalledApps()
    }

    private fun setupRecyclerView() {
        adapter = InstalledAppAdapter(
            apps = emptyList(),
            onCloneClick = { app ->
                cloneSelectedApp(app)
            }
        )
        rvInstalledApps.layoutManager = LinearLayoutManager(this)
        rvInstalledApps.adapter = adapter
    }

    private fun setupSearch() {
        etSearch.addTextChangedListener(object : TextWatcher {
            override fun beforeTextChanged(s: CharSequence?, start: Int, count: Int, after: Int) {}
            override fun onTextChanged(s: CharSequence?, start: Int, before: Int, count: Int) {
                filterApps(s.toString())
            }
            override fun afterTextChanged(s: Editable?) {}
        })
    }

    @SuppressLint("SetTextI18n")
    private fun filterApps(query: String) {
        val filtered = if (query.isBlank()) {
            allApps
        } else {
            allApps.filter {
                it.appName.contains(query, ignoreCase = true) ||
                it.packageName.contains(query, ignoreCase = true)
            }
        }
        txtAppCount.text = "${filtered.size} Aplikasi Ditemukan"
        adapter.updateData(filtered)
    }

    private fun loadInstalledApps() {
        progressBar.visibility = View.VISIBLE
        lifecycleScope.launch(Dispatchers.IO) {
            val pm = packageManager
            val packages = pm.getInstalledPackages(PackageManager.GET_META_DATA)
            val clonedPackages = appContainer.getClonedApps().map { it.packageName }.toSet()

            val list = mutableListOf<InstalledAppInfo>()
            for (pkg in packages) {
                val appInfo = pkg.applicationInfo ?: continue
                // Don't list sandbox itself
                if (pkg.packageName == packageName) continue

                val isSystem = (appInfo.flags and ApplicationInfo.FLAG_SYSTEM) != 0
                val appName = pm.getApplicationLabel(appInfo).toString()
                val versionName = pkg.versionName ?: "1.0"
                val versionCode = if (android.os.Build.VERSION.SDK_INT >= android.os.Build.VERSION_CODES.P) {
                    pkg.longVersionCode
                } else {
                    @Suppress("DEPRECATION")
                    pkg.versionCode.toLong()
                }

                list.add(
                    InstalledAppInfo(
                        packageName = pkg.packageName,
                        appName = appName,
                        versionName = versionName,
                        versionCode = versionCode,
                        sourceDir = appInfo.sourceDir,
                        isSystemApp = isSystem,
                        isAlreadyCloned = clonedPackages.contains(pkg.packageName)
                    )
                )
            }

            // Sort: non-system apps first, then alphabetically
            list.sortWith(compareBy({ it.isSystemApp }, { it.appName.lowercase() }))

            withContext(Dispatchers.Main) {
                allApps = list
                progressBar.visibility = View.GONE
                filterApps(etSearch.text.toString())
            }
        }
    }

    private fun cloneSelectedApp(app: InstalledAppInfo) {
        progressBar.visibility = View.VISIBLE
        lifecycleScope.launch(Dispatchers.IO) {
            try {
                val cloned = appContainer.cloneApp(app.packageName)
                withContext(Dispatchers.Main) {
                    progressBar.visibility = View.GONE
                    // PRD UI requirement: bottom toast notification pop-up without blocking dialog
                    StealthToast.show(
                        this@AppClonerActivity,
                        "Aplikasi ${cloned.appName} berhasil dikloning ke sandbox!"
                    )
                    finish()
                }
            } catch (e: Exception) {
                withContext(Dispatchers.Main) {
                    progressBar.visibility = View.GONE
                    StealthToast.show(
                        this@AppClonerActivity,
                        "Gagal mengkloning: ${e.message}"
                    )
                }
            }
        }
    }
}

    private fun checkForAppUpdate(silent: Boolean = false) {

        val progressBar = dialogView.findViewById<ProgressBar>(R.id.progress_update_download)
        val layoutButtons = dialogView.findViewById<LinearLayout>(R.id.layout_update_buttons)
        val btnUpdateNow = dialogView.findViewById<Button>(R.id.btn_update_now)
        val btnUpdateLater = dialogView.findViewById<Button>(R.id.btn_update_later)

        txtVersion.text = "Versi Baru: ${updateInfo.latestVersion} • Versi Anda: v${BuildConfig.VERSION_NAME}"
        txtNotes.text = updateInfo.releaseNotes.ifEmpty { "Pembaruan kode dan peningkatan stealth engine." }

        val dialog = AlertDialog.Builder(this)
            .setView(dialogView)
            .setCancelable(true)
            .create()

        dialog.window?.setBackgroundDrawableResource(android.R.color.transparent)

        btnUpdateLater.setOnClickListener {
            dialog.dismiss()
        }

        btnUpdateNow.setOnClickListener {
            btnUpdateNow.isEnabled = false
            btnUpdateLater.visibility = View.GONE
            layoutProgress.visibility = View.VISIBLE
            dialog.setCancelable(false)

            lifecycleScope.launch {
                val apkFile = UpdateManager.downloadApk(
                    context = this@MainActivity,
                    downloadUrl = updateInfo.downloadUrl,
                    onProgress = { percent ->
                        progressBar.progress = percent
                        txtProgress.text = "Mengunduh pembaruan: $percent%"
                    }
                )

                if (apkFile != null && apkFile.exists()) {
                    txtProgress.text = "Menginstal pembaruan..."
                    dialog.dismiss()
                    // Installs APK directly via FileProvider after integrity check (SEC-04)
                    UpdateManager.installApk(this@MainActivity, apkFile, updateInfo.sha256Checksum)
                } else {
                    txtProgress.text = "Gagal mengunduh pembaruan"
                    btnUpdateNow.isEnabled = true
                    btnUpdateLater.visibility = View.VISIBLE
                    dialog.setCancelable(true)
                    StealthToast.show(this@MainActivity, "Gagal mengunduh APK pembaruan")
                }
            }
        }

        dialog.show()
    }

package com.sandbox.ball.ui.view

import android.annotation.SuppressLint
import android.content.Context
import android.view.Gravity
import android.view.LayoutInflater
import android.widget.TextView
import android.widget.Toast
import com.sandbox.ball.R

object StealthToast {
    /**
     * PRD UI Requirement:
     * Menggunakan toast notification pop-up di bagian bawah layar saat profil
     * identitas berhasil direset atau aplikasi selesai dikloning, sehingga tidak ada
     * dialog box yang menghalangi layar.
     */
    @SuppressLint("InflateParams")
    fun show(context: Context, message: String, duration: Int = Toast.LENGTH_SHORT) {
        val toast = Toast(context)
        val inflater = LayoutInflater.from(context)
        val view = inflater.inflate(R.layout.custom_stealth_toast, null)
        val text = view.findViewById<TextView>(R.id.txt_toast_message)
        text.text = message

        toast.view = view
        toast.duration = duration
        toast.setGravity(Gravity.BOTTOM or Gravity.CENTER_HORIZONTAL, 0, 140)
        toast.show()
    }
}

class BinderProxyManagerTest {

    fun testPlayStoreInstallerSpoofing() {
        val installer = PackageManagerHook.getInstallerPackageName("com.miniclip.eightballpool")
        assertEquals("Installer must be Google Play Store", "com.android.vending", installer)

        val origin = PackageManagerHook.getOriginatingPackageName()
        assertEquals("Origin must be Google Play Store", "com.android.vending", origin)
    }

    @Test
    fun testPackageVisibilityFiltering() {
        // Sandboxed package must be visible
        PackageManagerHook.setTargetPackage("com.miniclip.eightballpool")
        assertTrue(PackageManagerHook.isPackageVisible("com.miniclip.eightballpool"))

        // Common system packages must be visible
        assertTrue(PackageManagerHook.isPackageVisible("android"))
        assertTrue(PackageManagerHook.isPackageVisible("com.google.android.gms"))
        assertTrue(PackageManagerHook.isPackageVisible("com.android.vending"))

        // Suspicious tool packages must be hidden
        assertFalse(PackageManagerHook.isPackageVisible("de.robv.android.xposed.installer"))
        assertFalse(PackageManagerHook.isPackageVisible("com.topjohnwu.magisk"))
        assertFalse(PackageManagerHook.isPackageVisible("com.noshufou.android.su"))

        // Sandbox host package must be hidden from guest game
        assertFalse(PackageManagerHook.isPackageVisible("com.sandbox.ball"))

        // When no guest app is running (host dashboard mode), host app must be visible to itself
        PackageManagerHook.setTargetPackage(null)
        assertTrue(PackageManagerHook.isPackageVisible("com.sandbox.ball"))
    }

    @Test
    fun testSettingsSecureIntercept() {
        val profile = HardwareProfile(androidId = "aabbccddeeff0011", model = "SM-S928B", brand = "samsung")
        SettingsSecureHook.setProfile(profile)

        val interceptedId = SettingsSecureHook.intercept(Settings.Secure.ANDROID_ID, "original_hardware_id")
        assertEquals("aabbccddeeff0011", interceptedId)

        val interceptedName = SettingsSecureHook.intercept("device_name", "original_device")
        assertEquals("SM-S928B", interceptedName)
    }

    @Test
    fun testWifiHookSpoofing() {
        val profile = HardwareProfile(

        assertEquals("02:11:22:33:44:55", WifiHook.getMacAddress())

package com.sandbox.ball

import com.sandbox.ball.core.spoofer.ProfileGenerator
import org.junit.Assert.*
import org.junit.Test

class ProfileGeneratorTest {

    @Test
    fun testGenerateLuhnImei() {
        val profile = ProfileGenerator.generateRandomProfile()
        val imei = profile.imei

        assertEquals("IMEI must be 15 digits long", 15, imei.length)
        assertTrue("IMEI must contain only digits", imei.all { it.isDigit() })

        // Verify Luhn checksum
        var sum = 0
        for (i in imei.indices) {
            var digit = imei[i] - '0'
            if (i % 2 == 1) {
                digit *= 2
                if (digit > 9) digit -= 9
            }
            sum += digit
        }
        assertEquals("IMEI Luhn checksum modulo 10 must be 0", 0, sum % 10)
    }

    @Test
    fun testGenerateAndroidId() {
        val profile = ProfileGenerator.generateRandomProfile()
        val androidId = profile.androidId

        assertEquals("Android ID must be 16 characters", 16, androidId.length)
        assertTrue("Android ID must be valid hex", androidId.matches(Regex("[0-9a-f]{16}")))
    }

    @Test
    fun testGenerateMacAddress() {
        val profile = ProfileGenerator.generateRandomProfile()
        val mac = profile.macAddress

        assertTrue("MAC address must match standard 6-byte hex format",
            mac.matches(Regex("^([0-9A-Fa-f]{2}[:-]){5}([0-9A-Fa-f]{2})$")))
    }

    @Test
    fun testDevicePresets() {
        val samsung = ProfileGenerator.generatePreset(ProfileGenerator.DevicePreset.SAMSUNG_S24_ULTRA)

    }

                fail("Expected SecurityException for nested sub-directory entry")

# Product Requirements Document (PRD): Stealth Sandbox Engine

**Status Dokumen:** Draft / Perencanaan
**Repositori:** `github.com/Dayat222/sanbox_ball`
**Target Platform:** Android (API 21 - API 34)

---

## 1. Latar Belakang & Ringkasan Eksekutif
Sistem *anti-cheat* modern pada aplikasi dan game telah bergeser dari sekadar mendeteksi modifikasi memori menjadi pelacakan identitas perangkat keras (Hardware ID). Ketika pelanggaran terdeteksi, server akan memblokir secara permanen pengenal fisik perangkat (IMEI, Android ID, MAC Address). 

**Stealth Sandbox Engine** dirancang sebagai lingkungan virtual terisolasi yang berfungsi sebagai tameng. Sistem ini mencegat setiap permintaan data identitas perangkat, memberikan data palsu, dan menyembunyikan status virtualisasinya.

## 2. Tujuan & Matriks Keberhasilan

### Tujuan
* **Isolasi Hardware:** Melindungi identitas asli perangkat dari pembacaan pihak ketiga.
* **Bypass Anti-Cheat:** Menghindari deteksi lingkungan virtual.
* **Efisiensi Manajemen:** Memberikan kontrol instan bagi pengguna untuk mereset identitas virtual dalam satu ketukan saat akun terblokir.

### Matriks Keberhasilan
* **0% Device Ban Rate:** Perangkat fisik utama tidak pernah masuk daftar hitam.
* **Crash-Free Rate > 95%:** Aplikasi target dapat dimuat di dalam sandbox tanpa force close.
* **Evasion Success:** Lolos dari pemeriksaan PackageManager dan deteksi ptrace.

---

## 3. Spesifikasi Fungsional (Core Features)

| Fitur | Deskripsi Fungsional | Prioritas |
| :--- | :--- | :--- |
| **Kloning Aplikasi** | Menyalin APK dari OS utama ke dalam kontainer tanpa root. | Tinggi |

## 4. Persyaratan Non-Fungsional (Anti-Detection & Evasion)

* **System Call Interception:** Menerapkan intersepsi di level native (C/C++) menggunakan inline hooking. VFS (Virtual File System) akan mengarahkan pembacaan file sistem sensitif ke file teks palsu.
* **Mount Namespace & Direktori:** Menyembunyikan path instalasi asli agar terlihat seperti direktori `/data/app/` standar.
* **Anti-Debugger Bypass:** Menyisipkan hook pada fungsi `ptrace` agar selalu mengembalikan nilai sukses palsu (`0`).

2. **Java Framework Middleware:** Mengelola isolasi Android Framework (mengganti Context, mencegat TelephonyManager).

* **Struktur Visual:** Daftar menu profil dan status aplikasi dibuat berpoin serta terstruktur rapi tanpa ornamen grafis yang berlebihan.

---

## 7. Fase Pengembangan (Milestones)

# Product Requirements Document (PRD): Perbaikan Arsitektur & DevSecOps (v1.1.0)

**Nama Proyek:** Stealth Sandbox Engine (`sanbox_ball`)  
**Status Dokumen:** Rencana Perbaikan Teknis (Remediation Plan)  
**Target Rilis:** v1.1.0-release  
**Platform Target:** Android API 21 $\rightarrow$ API 34 (Compile SDK 35)

---

## 1. Latar Belakang & Tujuan Perbaikan
Berdasarkan hasil audit teknis pada versi `1.0.3`, ditemukan bahwa antarmuka pengguna (UI) dan generator profil telah berfungsi dengan baik, namun lapisan inti kontainer (*container runtime*), pencegatan panggilan sistem (*hooking*), serta konfigurasi *build* masih bersifat pasif (simulasi) dan memiliki celah *thread-safety*.

Dokumen PRD ini mendefinisikan spesifikasi teknis untuk memperbaiki empat area kritis agar aplikasi dapat menjalankan kontainer secara terisolasi, aman dari kondisi *race condition*, dan terkompilasi sesuai standar produksi.

---

## 2. Ruang Lingkup Perbaikan (Remediation Scope)

| ID | Area Perbaikan | Masalah Saat Ini | Solusi Arsitektur Target | Prioritas |
| :--- | :--- | :--- | :--- | :--- |
| **REM-01** | **Container Runtime** | `StubActivity` membuka aplikasi via `Intent` eksternal di OS utama. | Pemuatan APK secara internal di dalam proses `:container` menggunakan refleksi `ActivityThread` dan `LoadedApk`. | Kritis |
| **REM-02** | **Native PLT/GOT Hooking** | `syscall_hook.cpp` hanya memanggil `dlsym` tanpa menimpa tabel simbol memori. | Implementasi *PLT/GOT hooking* untuk mengarahkan `open`, `openat`, `fopen`, `stat`, dan `access` ke fungsi pengalihan VFS. | Kritis |
| **REM-03** | **VFS Thread-Safety** | `sanitizeMapsFile()` dipanggil di luar kunci `std::mutex`, memicu *race condition*. | Enkapsulasi penuh dengan `std::lock_guard<std::mutex>` atau penggunaan deskriptor memori anonim (`memfd_create`). | Tinggi |
| **REM-04** | **Binder Proxy Hooking** | Kelas di `core/hooks/` bersifat statis pasif dan tidak terhubung ke sistem Android. | Implementasi Java `Dynamic Proxy` (`InvocationHandler`) pada `ServiceManager` untuk mencegat transaksi IPC Binder. | Kritis |
| **REM-05** | **NDK Build System** | File `CMakeLists.txt` belum terdefinisi secara eksplisit dalam paket *build*. | Pembuatan konfigurasi `src/main/cpp/CMakeLists.txt` untuk kompilasi `libstealth_engine.so`. | Tinggi |
| **REM-06** | **DevSecOps & CI/CD** | Rilis menggunakan `app-debug.apk`, `isMinifyEnabled = false`, dan `local.properties` bocor. | Kompilasi `assembleRelease` dengan R8/ProGuard aktif, *signing key* via GitHub Secrets, dan pembersihan `.gitignore`. | Tinggi |

---

## 3. Spesifikasi Teknis Perbaikan

### 3.1. Perbaikan Eksekusi Kontainer (`core/container/`)
Saat ini `StubActivity.kt` memanggil `packageManager.getLaunchIntentForPackage(pkg)` yang menyerahkan eksekusi kembali ke OS utama.
* **Kebutuhan Teknis:**
  1. **Context & LoadedApk Replacement:** Saat `StubActivity` dijalankan di proses `:container`, sistem harus membuat instance `Context` khusus yang mengarahkan `getDataDir()`, `getFilesDir()`, dan `getCacheDir()` ke direktori `VirtualFileSystem` (`/sandbox_vfs/data/<pkg>`).
  2. **Instrumentation & Activity Lifecycle:** Mengganti `mInstrumentation` pada `ActivityThread` saat ini agar setiap pemanggilan `execStartActivity` dari aplikasi yang dikloning dibungkus kembali ke dalam `StubActivity` milik kontainer.
  3. **Native Library Loading:** Memastikan direktori `lib/<abi>` dari `base.apk` diekstrak ke `/sandbox_vfs/apps/<pkg>/lib/` dan ditambahkan ke *path* pencarian `DexClassLoader`.

### 3.2. Perbaikan Lapisan Native C++ (`src/main/cpp/`)
#### A. Implementasi PLT/GOT Interception (`syscall_hook.cpp`)
* `dlsym(libc, "open")` hanya mengambil alamat fungsi asli. Untuk mengalihkan panggilan dari *library* yang dimuat di dalam kontainer, sistem harus memindai tabel PLT/GOT dari *shared libraries* (`.so`) yang dimuat di memori proses `:container` dan mengganti alamat fungsi `open`, `openat`, `fopen`, `stat`, `lstat`, dan `access` ke fungsi *wrapper* `hooked_*`.

#### B. Perbaikan Race Condition pada VFS (`vfs_core.cpp`)
* Pada metode `VfsCore::resolvePath(const char* originalPath)`:
  * Pindahkan kunci `std::lock_guard<std::mutex> lock(mutex_);` ke baris paling atas sebelum pemeriksaan `sanitizeMapsFile()`.
  * Gunakan penulisan file atomik (menulis ke file sementara `proc_maps.<tid>` lalu mengganti namanya, atau menggunakan sistem *in-memory file descriptor* `memfd_create("sanitized_maps", MFD_CLOEXEC)`) agar pembacaan `/proc/self/maps` oleh banyak *thread* sekaligus tidak saling menimpa.

#### C. Konfigurasi `CMakeLists.txt`
* Menyediakan file `app/src/main/cpp/CMakeLists.txt` yang mendefinisikan:
  * Target *shared library*: `add_library(stealth_engine SHARED stealth_engine.cpp vfs/vfs_core.cpp hook/syscall_hook.cpp)`
  * Flag visibilitas simbol: `-fvisibility=hidden` dan `-Oz` (optimasi ukuran biner pada mode rilis).
  * Penautan pustaka sistem: `target_link_libraries(stealth_engine android log dl)`.

### 3.3. Perbaikan Lapisan Middleware Framework (`core/hooks/`)
Agar objek `TelephonyHook`, `WifiHook`, `SettingsSecureHook`, dan `PackageManagerHook` benar-benar bekerja saat aplikasi berjalan:
* **Binder IPC Interception:**
  1. Membuat komponen `BinderProxyManager.kt` yang mengganti objek `IInterface` pada *cache* `ServiceManager` menggunakan `java.lang.reflect.Proxy.newProxyInstance`.
  2. **PackageManager (`IPackageManager`):** Mencegat metode `getInstalledPackages`, `getInstalledApplications`, `getPackageInfo`, dan `getInstallerPackageName`, lalu meneruskan hasilnya ke filter yang sudah ada di `PackageManagerHook.kt`.
  3. **Telephony (`ITelephony` & `IPhoneSubInfo`):** Mencegat metode `getDeviceId`, `getImeiForSlot`, dan `getSubscriberIdForSubscriber`, lalu mengembalikan nilai dari `TelephonyHook.kt`.
  4. **Settings Provider (`IContentProvider`):** Mencegat panggilan `CALL_METHOD` untuk `GET_secure` (`android_id`) dan mengembalikan nilai dari `SettingsSecureHook.kt`.
* **Kompatibilitas Android 9+ (`BuildPropsHook.kt`):**
  * Menghapus penggunaan refleksi `Field::class.java.getDeclaredField("modifiers")` yang diblokir oleh *Hidden API Policy* pada Android 9 $\rightarrow$ 14. Cukup gunakan `field.isAccessible = true` langsung pada *field* statis kelas `android.os.Build` atau lakukan modifikasi properti melalui jembatan JNI (`env->SetStaticObjectField`) di `stealth_engine.cpp` karena lapisan JNI memiliki akses lebih leluasa terhadap *field* `static final` dibandingkan refleksi Java.

---

## 4. Spesifikasi Keamanan Build & CI/CD (DevSecOps)

### 4.1. Pengamanan Konfigurasi Gradle (`app/build.gradle.kts`)
1. **Aktifkan Obfuscation (R8/ProGuard) pada Mode Release:**
   * Mengubah `isMinifyEnabled = true` dan menambahkan `isShrinkResources = true` pada blok `buildTypes.release`.
   * Menghapus simbol *debug* dari pustaka C++ dengan menambahkan `ndk { debugSymbolLevel = "NONE" }` pada blok `release`.
2. **Pembersihan File Lokal:**
   * Menghapus `local.properties` dari indeks Git (`git rm --cached local.properties`) dan menambahkannya ke dalam `.gitignore`.

### 4.2. Pembaruan Alur Kerja GitHub Actions (`.github/workflows/build.yml`)
1. Mengganti perintah `./gradlew assembleDebug` menjadi `./gradlew assembleRelease` untuk rilis produksi agar APK tidak memiliki atribut `android:debuggable="true"`.
2. Menambahkan langkah *APK Signing* atau konfigurasi `signingConfigs` di Gradle yang membaca variabel lingkungan dari GitHub Secrets (`KEYSTORE_BASE64`, `KEYSTORE_PASSWORD`, `KEY_ALIAS`, `KEY_PASSWORD`).
3. Memastikan `UpdateManager.kt` mencari aset APK rilis (`*-release.apk`) terlebih dahulu sebelum jatuh kembali (*fallback*) ke aset lainnya.

---

## 5. Kriteria Penerimaan (Acceptance Criteria) & Pengujian

1. **Verifikasi Isolasi Kontainer:**
   * Saat aplikasi target dijalankan melalui `StubActivity`, pemeriksaan `ps -A` atau `ActivityManager.getRunningAppProcesses()` harus menunjukkan bahwa aplikasi berjalan di dalam proses `com.sandbox.ball:container`, bukan sebagai proses mandiri di luar *sandbox*.
2. **Verifikasi Intersepsi Identitas & VFS:**
   * Unit test dan *instrumentation test* (`androidTest`) yang memanggil `Settings.Secure.getString(resolver, ANDROID_ID)` serta membaca `/sys/class/net/wlan0/address` melalui `FileInputStream` di dalam kontainer harus menerima nilai dari `HardwareProfile` yang sedang aktif, bukan nilai perangkat asli.
3. **Verifikasi Thread-Safety VFS:**
   * Pengujian pembacaan `/proc/self/maps` secara paralel oleh 10 *thread* bersamaan selama 5 detik tidak boleh menghasilkan *crash* (`SIGSEGV`/`SIGABRT`) atau *file descriptor leak*.
4. **Verifikasi Build Rilis:**
   * GitHub Actions berhasil menghasilkan APK rilis yang telah di-*obfuscate* oleh R8 dan ditandatangani secara valid tanpa menyertakan bendera `FLAG_DEBUGGABLE`.

Celah keamanan tersebut berpotensi dieksploitasi oleh aplikasi berbahaya yang dikloning untuk keluar dari isolasi kontainer (*Sandbox Escape*), menimpa berkas internal aplikasi *host* (*Zip Slip*), melakukan serangan *Man-in-the-Middle* (MitM) pada mekanisme pembaruan aplikasi (*In-App Updater*), serta memicu *Undefined Behavior* akibat *Data Race* pada lapisan C++.

Dokumen PRD ini menetapkan spesifikasi teknis tingkat kode untuk menutup seluruh celah keamanan tersebut agar arsitektur kontainer memenuhi standar keamanan produksi.

## 3. Spesifikasi Teknis Perbaikan Lapisan Kotlin/Java

### 3.1. Mitigasi Kerentanan *Zip Slip* pada Ekstraktor Native (`SEC-01`)
* **Lokasi Berkas:** `app/src/main/java/com/sandbox/ball/core/container/NativeLibExtractor.kt`
* **Analisis Masalah:**
  Saat mengekstrak berkas `.so` dari APK yang dikloning, kode mengambil `val soName = entry.name.substring(prefix.length)` lalu langsung membuka `FileOutputStream(File(targetLibDir, soName))`. APK yang dimodifikasi dengan nama entri `lib/arm64-v8a/../../../files/exploit.so` dapat menimpa berkas di luar direktori target.
* **Spesifikasi Perbaikan:**
  1. Ambil jalur kanonik dari direktori tujuan terlebih dahulu: `val canonicalDestDir = targetLibDir.canonicalPath + File.separator`.
  2. Tolak setiap nama entri yang mengandung karakter pemisah direktori terlarang atau yang jalur kanoniknya keluar dari `canonicalDestDir`:
     ```kotlin
     val soName = entry.name.substring(prefix.length)
     if (soName.contains("/") || soName.contains("\\") || soName.contains("..")) {
         Log.w(TAG, "Blocked suspicious zip entry name: ${entry.name}")
         continue
     }

     val outFile = File(targetLibDir, soName)
     val canonicalOutPath = outFile.canonicalPath
     if (!canonicalOutPath.startsWith(canonicalDestDir)) {
         throw SecurityException("Zip Slip attack detected on entry: ${entry.name}")
     }
     ```

### 3.2. Pengamanan *In-App Updater* & Verifikasi Integritas APK (`SEC-04`)
* **Lokasi Berkas:** `app/src/main/java/com/sandbox/ball/core/updater/UpdateManager.kt` & `AppUpdateInfo.kt`
* **Analisis Masalah:**
  1. `UpdateManager.downloadApk()` mengikuti *redirect* HTTP (`301/302/307/308`) tanpa memeriksa apakah URL tujuan (`Location`) diturunkan ke protokol tidak terenkripsi (`http://`) atau diarahkan ke domain berbahaya.
  2. Berkas APK disimpan di `context.getExternalFilesDir(null)` yang lebih rentan terhadap modifikasi pihak ketiga dibandingkan ruang penyimpanan internal aplikasi.
  3. Tidak ada verifikasi *checksum* SHA-256 maupun pemeriksaan kesesuaian tanda tangan kriptografi (*APK Signature Verification*) sebelum memanggil `installApk()`.
* **Spesifikasi Perbaikan:**
  1. **Validasi Protokol & Domain pada Redirect:**
     * Buat fungsi pembantu `isTrustedUpdateUrl(url: String): Boolean` yang memastikan URL menggunakan skema `https://` dan berasal dari domain resmi GitHub (`github.com` atau `objects.githubusercontent.com`).
     * Sebelum mengikuti header `Location`, panggil `isTrustedUpdateUrl(newUrl)`. Jika tidak valid, batalkan pengunduhan dan lempar `SecurityException`.
  2. **Isolasi Penyimpanan Unduhan:**
     * Ubah lokasi penyimpanan berkas unduhan dari `context.getExternalFilesDir(null)` menjadi direktori internal `File(context.cacheDir, "updates")`.
  3. **Verifikasi SHA-256 & Sertifikat Penandatangan APK:**
     * Tambahkan properti `val sha256Checksum: String = ""` pada `AppUpdateInfo.kt` (diekstrak dari catatan rilis GitHub jika tersedia, misalnya format `SHA256: <64-hex>`).
     * Buat metode `verifyDownloadedApk(context: Context, apkFile: File, expectedSha256: String): Boolean` di `UpdateManager.kt`:
       * Jika `expectedSha256` tidak kosong, hitung digest `SHA-256` dari `apkFile` dan bandingkan secara *constant-time* (`MessageDigest.isEqual`).
       * Gunakan `PackageManager.getPackageArchiveInfo()` dengan flag `GET_SIGNING_CERTIFICATES` (API 28+) atau `GET_SIGNATURES` (API 21–27) untuk memverifikasi bahwa nama paket di dalam APK yang diunduh adalah `context.packageName` dan sertifikat penandatangannya identik dengan sertifikat aplikasi yang sedang terinstal.
       * Jika verifikasi gagal, hapus `apkFile` seketika dan batalkan proses instalasi.

## 4. Spesifikasi Teknis Perbaikan Lapisan Native C++ (NDK)

### 4.1. Sanitasi *Path Traversal* & Perbaikan Pencocokan VFS (`SEC-02`)
* **Lokasi Berkas:** `app/src/main/cpp/vfs/vfs_core.cpp` & `app/src/main/cpp/vfs/vfs_core.h`
* **Analisis Masalah:**
  1. Pemanggilan `open("/data/data/<targetPkg>/../../../data/com.sandbox.ball/shared_prefs/...")` lolos dari pemeriksaan prefix dan memungkinkan aplikasi di dalam kontainer mengakses direktori milik *host* `com.sandbox.ball`.
  2. Pemeriksaan `pathStr.find("/maps") != std::string::npos` terlalu luas karena mencegat semua berkas aplikasi yang kebetulan memiliki kata `/maps` di dalamnya.
  3. Pencocokan prefix `pathStr.rfind(prefix, 0) == 0` tidak memeriksa karakter pembatas `/`, sehingga `/data/data/com.app` dapat mencocokkan `/data/data/com.app_other`.
* **Spesifikasi Perbaikan:**
  1. **Fungsi Normalisasi Jalur (`normalizeLexicalPath`):**
     Tambahkan fungsi di `vfs_core.cpp` yang memecah jalur berdasarkan `/` dan menyelesaikan setiap segmen `.` serta `..` secara leksikal sebelum pencocokan dilakukan:
     ```cpp
     static std::string normalizeLexicalPath(const std::string& rawPath) {
         if (rawPath.empty()) return "";
         bool isAbsolute = (rawPath[0] == '/');
         std::vector<std::string> parts;
         std::stringstream ss(rawPath);
         std::string item;
         while (std::getline(ss, item, '/')) {
             if (item.empty() || item == ".") continue;
             if (item == "..") {
                 if (!parts.empty()) parts.pop_back();
             } else {
                 parts.push_back(item);
             }
         }
         std::string result = isAbsolute ? "/" : "";
         for (size_t i = 0; i < parts.size(); ++i) {
             result += parts[i];
             if (i + 1 < parts.size()) result += "/";
         }
         return result;
     }
     ```
  2. **Proteksi Direktori Host Sandbox:**
     Setelah jalur dinormalisasi, apabila jalur mengarah ke direktori internal `com.sandbox.ball` di luar `virtualRootDir_` (misalnya `/data/data/com.sandbox.ball/shared_prefs` atau `/data/user/0/com.sandbox.ball/shared_prefs`), blokir akses dengan mengarahkannya ke `/dev/null` atau mengembalikan jalur tidak valid agar aplikasi target tidak dapat mencuri konfigurasi *host*.
  3. **Perbaikan Intersepsi `/proc/.../maps`:**
     Buat fungsi spesifik `isProcMapsPath(const std::string& path)`:
     ```cpp
     static bool isProcMapsPath(const std::string& path) {
         if (path == "/proc/self/maps" || path == "/proc/thread-self/maps") return true;
         if (path.rfind("/proc/", 0) == 0 && path.length() > 11) {
             return path.compare(path.length() - 5, 5, "/maps") == 0;
         }
         return false;
     }
     ```
  4. **Validasi Batas Direktori pada Prefix Mapping:**
     Pastikan karakter setelah prefix adalah `/` atau akhir dari string:
     ```cpp
     if (normPath.rfind(prefix, 0) == 0 &&
         (normPath.length() == prefix.length() || normPath[prefix.length()] == '/')) {
         std::string sub = normPath.substr(prefix.length());
         return pair.second + sub;
     }
     ```

### 4.2. Pemulihan Proteksi Memori GOT / RELRO (`SEC-03`)
* **Lokasi Berkas:** `app/src/main/cpp/hook/syscall_hook.cpp`
* **Analisis Masalah:**
  Pada `plt_hook_callback`, halaman memori `.got` diubah menggunakan `mprotect(page_start, page_size, PROT_READ | PROT_WRITE)` namun tidak dikembalikan ke izin semula. Apabila pustaka dikompilasi dengan *Full RELRO* (`GNU_RELRO`), membiarkan halaman GOT berstatus `RW` membuka celah *GOT Overwrite* dan memicu deteksi *anti-cheat* yang memindai adanya izin `w` pada segmen konstanta/RELRO di `/proc/self/maps`.
* **Spesifikasi Perbaikan:**
  1. Periksa apakah alamat `got_entry` berada di dalam segmen `PT_GNU_RELRO` milik objek ELF terkait (`phdr[i].p_type == PT_GNU_RELRO`).
  2. Jika `got_entry` berada di dalam rentang `PT_GNU_RELRO`, maka setelah nilai `*got_entry = target.hook_func` ditulis dan `__builtin___clear_cache` dipanggil, kembalikan izin halaman memori tersebut menjadi `PROT_READ`:
     ```cpp
     if (is_relro_page) {
         mprotect(page_start, page_size, PROT_READ);
     }
     ```
  3. Jika tidak berada di dalam `PT_GNU_RELRO` (berbagi halaman dengan `.data`), biarkan `PROT_READ | PROT_WRITE` agar tidak memicu `SIGSEGV`.

### 4.3. Perbaikan *Data Race* dengan `std::atomic` (`SEC-05`)
* **Lokasi Berkas:** `app/src/main/cpp/vfs/vfs_core.h`, `app/src/main/cpp/vfs/vfs_core.cpp`, dan `app/src/main/cpp/hook/syscall_hook.cpp`
* **Spesifikasi Perbaikan:**
  1. Di `vfs_core.h`:
     * Tambahkan `#include <atomic>`.
     * Ubah `bool enabled_ = false;` menjadi `std::atomic<bool> enabled_{false};`.
     * Ubah `bool mapsFilterEnabled_ = true;` menjadi `std::atomic<bool> mapsFilterEnabled_{true};`.
     * Gunakan `enabled_.load(std::memory_order_acquire)` pada pembacaan dan `enabled_.store(enabled, std::memory_order_release)` pada penulisan.
  2. Di `syscall_hook.cpp`:
     * Tambahkan `#include <atomic>`.
     * Ubah `static int g_antidebug_enabled = 1;` menjadi `static std::atomic<int> g_antidebug_enabled{1};`.
     * Ubah `static int g_hooks_installed = 0;` menjadi `static std::atomic<int> g_hooks_installed{0};`.

## 1. Ringkasan Eksekutif & Latar Belakang Masalah
Pada iterasi `v1.1.0-release`, proyek **Stealth Sandbox Engine** telah berhasil mengimplementasikan struktur dasar *PLT/GOT hooking*, sanitasi berkas `/proc/self/maps` berbasis ID *thread* (`gettid()`), pemalsuan properti `android.os.Build` melalui JNI, serta *pipeline* kompilasi rilis di GitHub Actions.

Namun, hasil audit kode secara menyeluruh terhadap `v1.1.0-release` menemukan **6 masalah teknis kritis** yang menyebabkan aplikasi berisiko gagal dikompilasi (*build failure*), mengalami *crash* saat dijalankan (`UnsatisfiedLinkError` dan `SIGSEGV`), serta gagal mengisolasi aplikasi target di dalam proses `:container`. Dokumen PRD lengkap ini merinci spesifikasi perbaikan tingkat kode untuk setiap komponen agar siap digunakan secara penuh.

## 3. Spesifikasi Teknis Perbaikan Lapisan Native C++ (NDK)

### 3.1. Pembuatan Konfigurasi Build Native (`FIX-01`)
* **Lokasi Berkas:** `app/src/main/cpp/CMakeLists.txt`
* **Spesifikasi:**
  * Menggunakan `cmake_minimum_required(VERSION 3.22.1)`.
  * Mendefinisikan pustaka bersama (*shared library*) bernama `stealth_engine`.
  * Mendaftarkan tiga berkas sumber utama:
    1. `stealth_engine.cpp`
    2. `vfs/vfs_core.cpp`
    3. `hook/syscall_hook.cpp`
  * Mengatur direktori *header* (`target_include_directories`) untuk `.`, `vfs`, `hook`, dan `utils`.
  * Menambahkan *compiler flags* `-fvisibility=hidden`, `-Wall`, `-Wextra`, serta `-Oz` untuk meminimalkan ukuran biner dan menyembunyikan simbol internal dari pemindaian tabel simbol.
  * Menautkan pustaka sistem Android: `android`, `log`, dan `dl`.

### 3.2. Perbaikan Deklarasi Simbol JNI & C++ Name Mangling (`FIX-02`)
* **Lokasi Berkas:** `app/src/main/cpp/stealth_engine.h` & `app/src/main/cpp/stealth_engine.cpp`
* **Analisis Akar Masalah:**
  Pada `v1.1.0`, fungsi `Java_com_sandbox_ball_core_nativebridge_NativeBridge_nativeApplyPltHooks` dan `Java_com_sandbox_ball_core_nativebridge_NativeBridge_nativeSetBuildProp` ditulis di bagian bawah `stealth_engine.cpp` tanpa deklarasi di dalam blok `extern "C"` pada `stealth_engine.h`. Kompiler Clang C++ akan mengubah nama simbol fungsi tersebut menjadi format *mangled* (contoh: `_Z69Java_com_sandbox...`), sehingga mesin virtual ART Android gagal menemukannya saat dipanggil dari `NativeBridge.kt`.
* **Spesifikasi Perbaikan:**
  1. Tambahkan deklarasi prototipe kedua fungsi tersebut di dalam blok `extern "C" { ... }` pada `stealth_engine.h`:
     ```cpp
     JNIEXPORT jboolean JNICALL
     Java_com_sandbox_ball_core_nativebridge_NativeBridge_nativeApplyPltHooks(
             JNIEnv* env, jobject thiz);

     JNIEXPORT jboolean JNICALL
     Java_com_sandbox_ball_core_nativebridge_NativeBridge_nativeSetBuildProp(
             JNIEnv* env, jobject thiz, jstring fieldName, jstring fieldValue);
     ```
  2. Bungkus seluruh implementasi fungsi JNI di dalam `stealth_engine.cpp` dengan blok `extern "C" { ... }` untuk menjamin tidak ada fungsi JNI yang terlewat di masa mendatang.

### 3.3. Perbaikan Proteksi Memori PLT/GOT Hooking (`FIX-03`)
* **Lokasi Berkas:** `app/src/main/cpp/hook/syscall_hook.cpp`
* **Analisis Akar Masalah:**
  1. Setelah mengganti nilai `*got_entry = target.hook_func;`, kode memanggil `mprotect(page_start, page_size, PROT_READ);`. Jika halaman memori (4096 byte) tempat `.got.plt` berada juga memuat variabel global yang dapat ditulis (`.data` atau `.bss`), pencabutan izin `PROT_WRITE` akan memicu `SIGSEGV` (*Segmentation Fault*) seketika saat pustaka menulis ke variabel globalnya.
  2. Perhitungan alamat tabel simbol pada beberapa arsitektur Android (khususnya Android API 21–23 vs API 24+) dapat memiliki `d->d_un.d_ptr` yang sudah berupa alamat absolut atau masih berupa *offset* relatif terhadap `info->dlpi_addr`.
* **Spesifikasi Perbaikan:**
  1. Setelah memodifikasi `*got_entry`, biarkan halaman memori tetap memiliki izin `PROT_READ | PROT_WRITE` atau panggil `__builtin___clear_cache` untuk menyinkronkan *instruction/data cache* CPU:
     ```cpp
     if (mprotect(page_start, page_size, PROT_READ | PROT_WRITE) == 0) {
         if (target.orig_func_ptr && !*(target.orig_func_ptr)) {
             *(target.orig_func_ptr) = *got_entry;
         }
         *got_entry = target.hook_func;
         __builtin___clear_cache(
             reinterpret_cast<char*>(got_entry),
             reinterpret_cast<char*>(got_entry) + sizeof(void*)
         );
     }
     ```
  2. Tambahkan juga intersepsi untuk simbol fungsi varian 64-bit di Android (`open64`, `openat64`, `fopen64`, `stat64`, `lstat64`) yang diarahkan ke *handler* yang sama, karena NDK modern sering memetakan panggilan file I/O ke varian 64-bit secara otomatis.

### 3.4. Pembersihan Berkas Sementara VFS (`FIX-06`)
* **Lokasi Berkas:** `app/src/main/cpp/vfs/vfs_core.cpp`
* **Spesifikasi Perbaikan:**
  * Pada `VfsCore::init()`, bersihkan seluruh berkas `proc_maps.<tid>` sisa sesi sebelumnya di dalam direktori `fake_sys` menggunakan `opendir`/`readdir`/`unlink` agar penyimpanan tidak penuh setelah penggunaan berulang.
  * Pada `VfsCore::shouldRedirect()`, pindahkan `std::lock_guard<std::mutex> lock(mutex_);` ke baris paling atas sebelum membaca `mapsFilterEnabled_` agar pembacaan status konsisten di semua *thread*.

## 4. Spesifikasi Teknis Perbaikan Container Runtime (`FIX-04`)

### 4.1. Penyempurnaan Injeksi Konteks & Pemuatan Activity (`StubActivity.kt` & `ContainerInstrumentation.kt`)
* **Analisis Akar Masalah:**
  Pada `v1.1.0`, ketika tombol `btn_container_launch` ditekan di `StubActivity.kt`, kode memanggil:
  ```kotlin
  val launchIntent = packageManager.getLaunchIntentForPackage(pkg)
  startActivity(launchIntent)
  ```
  Karena `ContainerInstrumentation.execStartActivity` hanya meneruskan `intent` tersebut ke `base` tanpa memodifikasi komponen target, sistem operasi Android membuka aplikasi asli di luar kontainer.
* **Spesifikasi Perbaikan:**
  1. **Resolusi Activity Utama dari APK Kontainer (`StubActivity.kt`):**
     * Gunakan `packageManager.getPackageArchiveInfo(apkPath, PackageManager.GET_ACTIVITIES)` untuk membaca daftar `ActivityInfo` langsung dari berkas `base.apk` yang ada di dalam kontainer (`/sandbox_vfs/apps/<pkg>/base.apk`).
     * Ambil nama kelas `Activity` utama (atau dari `getLaunchIntentForPackage(pkg)?.component?.className`).
  2. **Peluncuran Internal di Proses `:container`:**
     * Buat kelas `ContainerHostActivity` (atau gunakan mode *dynamic view hosting* di dalam `StubActivity`) yang berjalan di proses `:container`.
     * Di dalam `ContainerInstrumentation.kt`:
       * Simpan referensi ke `ContainerContext`, `DexClassLoader`, dan nama paket target yang sedang aktif.
       * Saat `execStartActivity` menangkap `Intent` yang menuju ke paket target (`intent.component?.packageName == activeTargetPkg`), simpan nama kelas `Activity` asli ke dalam `intent.putExtra("_stealth_real_activity", realClassName)` dan arahkan komponen `Intent` ke `StubActivity` / `ContainerHostActivity` agar tetap berada di dalam proses `:container`.
       * Override metode `newActivity(cl: ClassLoader?, className: String?, intent: Intent?)`: jika `intent` memiliki atribut `_stealth_real_activity`, muat kelas `Activity` tersebut menggunakan `containerClassLoader.loadClass(realActivityName).newInstance() as Activity`.
       * Override metode `callActivityOnCreate(activity: Activity?, icicle: Bundle?)`: sebelum memanggil `super.callActivityOnCreate`, suntikkan `ContainerContext` sebagai `mBase` dari `activity` menggunakan refleksi pada kelas `ContextWrapper` (`mBaseField.set(activity, containerContext)`), serta sesuaikan `mResources` pada `ContextThemeWrapper` agar sumber daya visual (`Resources` & `AssetManager`) milik aplikasi target termuat dengan benar.

## 1. Ringkasan Eksekutif & Latar Belakang

Sistem *anti-cheat* dan pelacak integritas perangkat modern pada aplikasi/game telah beralih dari deteksi modifikasi memori ke pemblokiran permanen pengenal fisik perangkat (**Hardware ID Ban** seperti IMEI, Android ID, MAC Address, dan Serial). 

**Stealth Sandbox Engine** dirancang sebagai lingkungan virtual terisolasi tanpa butuh akses root (*rootless container*) yang bertindak sebagai tameng. Sistem ini mencegat setiap permintaan data identitas perangkat, menyajikan data palsu yang realistis, serta menyembunyikan status virtualisasinya dari pemindai anti-cheat dan modul pendeteksi sandbox.

## 3. Arsitektur Anti-Detection & Native Evasion

### A. System Call Interception (C++ Native Hooker)
Intersepsi fungsi libc pada level low-level native untuk memotong akses langsung yang mencoba membaca informasi kernel/OS:
* `open` & `openat`: Meredireksi pembacaan file sistem sensitif.
* `access`, `stat`, `lstat`: Memastikan status file virtual valid.
* `fopen`: Meredireksi panggilan stream tingkat tinggi.

### B. Virtual File System (VFS) & Mount Namespace
* `/proc/cpuinfo` $\rightarrow$ Mengembalikan data prosesor standar (Qualcomm SM8550 Snapdragon 8 Gen 2 / Gen 3).
* `/proc/version` $\rightarrow$ Mengembalikan kernel release Android standar.
* `/sys/class/net/wlan0/address` $\rightarrow$ Mengembalikan alamat MAC tiruan.
* `/proc/self/maps` $\rightarrow$ **Sanitizer Engine** otomatis menyaring dan menghapus baris memori yang memuat library sandbox (`libstealth_engine.so`), package name (`com.sandbox.ball`), frida, dan xposed.
* Mount namespace mengarahkan query `/data/data/<pkg>` dan `/data/user/0/<pkg>` langsung ke kontainer terisolasi.

### C. Anti-Debugger Bypass (`ptrace`)
* Modul native menyisipkan hook pada fungsi libc `ptrace`.
* Permintaan anti-cheat seperti `ptrace(PTRACE_TRACEME, 0, 1, 0)` dan `ptrace(PTRACE_ATTACH, ...)` akan selalu dicegat dan mengembalikan nilai sukses palsu (`0`).
* Mencegah crash paksa atau suicide loop aplikasi target yang memeriksa debugger.
* File `/proc/self/status` dipalsukan dengan `TracerPid: 0`.

### D. Dynamic Package Renaming
* Menyamarkan nama paket internal dan menyaring referensi `com.sandbox.ball` agar scanner anti-cheat tidak dapat menemukan tanda-tanda aplikasi sandbox di memori maupun daftar proses.
